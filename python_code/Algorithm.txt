Q : shape (N, N), input, integers, could be floating point values ?

ds_p : shape(N), 0/1, starts with all zeros 
ds_n : shape(N), 0/1, starts with all zeros 

s_p : shape(N), 0/1, starts with all ones
s_n : shape(N), 0/1, starts with all zeros

vmem : Q @ (s_p - s_n), needs to be computed ! 

curr_cuts : starts with zeros ( can be computed as sum [ multiply [ Q , 1.0 - (s_p - s_n).T @ (s_p - s_n)  ] ] )

constants : 
- thld_max = 8000 (8 * 10^4)
- delta_thld = 0.002 (2 * 10^-3)
- MAX_ITER = 10^8
- alpha = 2.5e4
- beta = 2.5

variables :
- thld = 1

while (iter < MAX_ITER)
{
    # iterative 
    pick random neuron p from range [0, N] and save it as p_idx;

    # generate noisy_threshold 
    noise_thld = alpha / thld_max * log(beta * random_uniform(0, 1) + epsilon) / (log(thld + thld_max) - log(thld_max));
    
    // if (noise_thld - vmem[p_idx] * (s_p[p_idx] - s_n[p_idx]) ) < 0
    // if noise_thld < vmem[p_idx] * (s_p[p_idx] - s_n[p_idx])
    // if noise_thld < s_p[p_idx] ? vmem[p_idx] : -vmem[p_idx]
    if (noise_thld * (s_p[p_idx] - s_n[p_idx])  - vmem[p_idx]) * (s_p[p_idx] - s_n[p_idx]) < 0
        spike = 1;
    else
        spike = 0;


    if spike:
        ds_p[p_idx] = s_n[p_idx] > 0
        ds_n[p_idx] = s_p[p_idx] > 0

        # update the neuron states 
        s_p[p] = ~s_p[p_idx];
        s_n[p] = ~s_n[p_idx];

        # update neuron membrane potential 
        vmem = vmem + 2 * (ds_p[p_idx] - ds_n[p_idx]) * Q[p, :]

        # update best cuts 
        delta_cuts = (ds_p[p_idx] - ds_n[p_idx]) * vmem[p_idx]

        if delta_cuts > 0:
            # bad flip !
            bad_flips += 1
        
        curr_cuts -= delta_cuts

        best_maxcuts = max(best_maxcuts, curr_cuts)
        

    thld = thld + delta_thld
    iter += 1;
}



U =   Q[0][0] + Q[0][1] + Q[0][2] ... + Q[0][N]
U' = -Q[0][0] + Q[0][1] + Q[0][2] ... + Q[0][N]

U' = U - 2 * Q[0][0]
